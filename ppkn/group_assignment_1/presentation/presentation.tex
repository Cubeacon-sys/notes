\begin{frame}{Pendahuluan: Pancasila sebagai Jiwa Bangsa}
    \begin{itemize}
        \item \textbf{Bukti Nyata di Masyarakat:} Pengalaman bangsa Indonesia saat menghadapi pandemi Covid-19 membuktikan kuatnya semangat gotong royong dan kerja sama semua pihak.
        \item \textbf{Penyatuan Nilai:} Keterlibatan aktif masyarakat dalam menyelesaikan masalah bangsa menegaskan bahwa nilai-nilai Pancasila telah menyatu sebagai jiwa bangsa.
        \item \textbf{Karakter Bangsa:} Pembangunan karakter bangsa yang tangguh bertumpu pada kebiasaan sehat, komitmen, dan kerja sama yang berlandaskan sila-sila Pancasila.
        \item \textbf{Dinamika Praktis:} Penerapan dasar negara berkaitan erat dengan dinamika pelaksanaannya dalam sikap serta tindakan nyata sehari-hari.
    \end{itemize}
\end{frame}

\begin{frame}{Sejarah Rumusan Dasar Negara}
    \begin{itemize}
        \item \textbf{Sidang Pertama BPUPK (29 Mei 1945):} Ketua BPUPK, Radjiman Wedyodiningrat, memberikan pertanyaan pemantik mengenai apa landasan bagi negara Indonesia merdeka yang akan dibangun.
        \item \textbf{Panitia Delapan (Panitia Kecil):} Dibentuk pada 1 Juni 1945 yang diketuai oleh Ir. Sukarno, bertugas menampung serta memeriksa usulan-usulan rumusan dasar negara dari para anggota.
        \item \textbf{Panitia Sembilan:} Dibentuk sebagai langkah lanjutan untuk menyusun rancangan Pembukaan UUD. Kepanitiaan ini krusial dalam mempertemukan dua pandangan besar (golongan kebangsaan dan golongan Islam) mengenai dasar kenegaraan.
    \end{itemize}
\end{frame}

\begin{frame}{Piagam Jakarta dan Dinamika Perdebatan}
    \begin{itemize}
        \item \textbf{Lahirnya Piagam Jakarta:} Pada 22 Juni 1945, Panitia Sembilan menyepakati rancangan Pembukaan UUD yang memuat rumusan Pancasila.
        \item \textbf{Sidang BPUPK Kedua (10–17 Juli 1945):} Hasil rumusan tersebut memicu perdebatan, khususnya terkait pencantuman "tujuh kata" pada Sila Ketuhanan.
        \item \textbf{Titik Keberatan:} Sebagian golongan kebangsaan dan beberapa tokoh Islam merasa perlakuan khusus tersebut kurang relevan jika dimasukkan dalam hukum dasar yang mengikat seluruh warga negara.
        \item \textbf{Penengah Perdebatan:} Tanggapan keberatan dari J. Latuharhary pada 11 Juli 1945 sempat menghangatkan suasana, namun berhasil ditengahi untuk sementara waktu berkat kewibawaan Ir. Sukarno.
    \end{itemize}
\end{frame}

\begin{frame}{Kedudukan Pancasila Pasca-Kemerdekaan}
    Pada 7 Agustus 1945, BPUPK dibubarkan karena telah menyelesaikan tugas utamanya, yaitu menyusun norma dasar (\textit{grundnorm}) negara. Sejak 18 Agustus 1945, Pancasila disahkan dengan dua kedudukan krusial:
    \vspace{0.5cm}
    \begin{itemize}
        \item \textbf{\textit{Philosofische Grondslag} (Dasar Filsafat Negara):} \\ Mengandung argumen dan alasan filosofis berdirinya suatu negara. Segala produk hukum di Indonesia wajib bersumber dan berlandaskan pada nilai-nilai Pancasila.
        \item \textbf{\textit{Weltanschauung} (Pandangan Hidup Bangsa):} \\ Menjadi pedoman bermasyarakat karena sarat akan unsur-unsur nilai agama, kebudayaan, serta kearifan adat istiadat nusantara.
    \end{itemize}
\end{frame}

\begin{frame}{Nilai-Nilai Luhur Para Pendiri Bangsa}
    Keberhasilan merumuskan dasar negara tidak lepas dari keteladanan para tokoh bangsa:
    \begin{itemize}
        \item \textbf{Menghargai Pendapat \& Rendah Hati:} Menjunjung tinggi musyawarah, tidak memaksakan kehendak, dan legawa menerima gagasan yang lebih relevan untuk negara.
        \item \textbf{Kerja Keras \& Keberanian:} Mencurahkan segenap tenaga dan pikiran secara gigih, serta berani mengambil risiko politik di bawah tekanan pemerintah militer Jepang.
        \item \textbf{Rela Berkorban:} Mengorbankan waktu, pikiran, hingga materi sebagai bentuk bakti tanpa pamrih kepada republik.
        \item \textbf{Mengutamakan Persatuan:} Mampu mengesampingkan ego kelompok demi keutuhan bangsa, tercermin dari kesepakatan mengubah redaksi Sila Pertama pada Piagam Jakarta.
    \end{itemize}
\end{frame}
