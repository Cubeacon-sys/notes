\section{Pancasila: Asal Mula \& Nilai}

% SLIDE 3
\begin{frame}{Pancasila: Dari Mana \& Untuk Apa}
    \begin{itemize}
        \item Pancasila = \textbf{pandangan hidup bangsa}, sudah ada jauh sebelum merdeka
        (\textit{Bhinneka Tunggal Ika} $\to$ Kitab Sutasoma, Majapahit).
        \item \textbf{4 Asal Mula (Kaelan, 2020)}: Bahan (digali dari adat, budaya, nilai
        religius) $\to$ Bentuk (dirumuskan BPUPK, disahkan PPKI) $\to$ Karya (PPKI sebagai
        pembentuk negara yang mengesahkan Pancasila jadi dasar negara sah) $\to$ Tujuan
        (dirumuskan pendiri negara untuk ditetapkan sebagai dasar negara).
        \item \textbf{3 Tingkatan Nilai}: Nilai Dasar (fundamental \& universal) $\to$ Nilai
        Instrumental $\to$ Nilai Praksis (penerapan nyata).
    \end{itemize}
\end{frame}

% SLIDE 9
\begin{frame}{Amendemen 1999--2002: 5 Kesepakatan \& Perubahan}
    \footnotesize
    \textbf{5 Kesepakatan Dasar:} (1) Pembukaan tidak diubah (2) NKRI tetap (3) Sistem
    presidensial dipertegas (4) Penjelasan $\to$ masuk pasal (5) Diubah via \textit{addendum}.
    \vspace{2mm}

    \textbf{Perubahan Struktural Kunci:}
    \begin{itemize}
        \item Sistematika: (Pembukaan, Batang Tubuh, Penjelasan) $\to$ (Pembukaan,
        Pasal-Pasal).
        \item Presiden: dipilih MPR $\to$ dipilih \textbf{langsung oleh rakyat}.
        \item Kedaulatan: di tangan MPR $\to$ di tangan \textbf{rakyat} menurut UUD.
        \item DPA dihapus; lahir lembaga baru: \textbf{MK, KY, DPD}.
    \end{itemize}
\end{frame}

\section{Hak \& Kewajiban Warga Negara}

% SLIDE 11
\begin{frame}{Hak \& Kewajiban: Konsep dan Contoh}
    \small \textbf{Timbal balik (\textit{reciprocity})}: hak dan kewajiban tidak bisa
    dipisah, harus seimbang. Pasal kunci: \textbf{27} (hukum \& pekerjaan), \textbf{28A-J}
    (HAM), \textbf{29} (agama), \textbf{30} (bela negara), \textbf{31-34} (pendidikan,
    ekonomi, sosial).
    \vspace{2mm}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{block}{\centering Contoh Hak}
                \footnotesize Beragama $\cdot$ Berpendapat $\cdot$ Pendidikan $\cdot$ Milik
                properti $\cdot$ Sama di mata hukum
            \end{block}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{block}{\centering Contoh Kewajiban}
                \footnotesize Bayar pajak $\cdot$ Taat hukum $\cdot$ Hormati hak orang lain
                $\cdot$ Jaga fasilitas umum
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
