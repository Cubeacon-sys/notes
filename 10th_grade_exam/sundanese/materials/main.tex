\section{METODOLOGI JEUNG ETIKA WAWANCARA}
% ==============================================================

\subsection{Definisi jeung Esensi Komunikasi Wawancara}
Sacara harti dasar, \textbf{tanya jawab langsung pikeun nyaring informasi ti narasumber disebutna wawancara}. Ieu kagiatan téh mangrupa prosés interaksi verbal (paguneman) antara dua urang atawa leuwih anu boga tujuan pikeun meunangkeun katerangan (informasi) ti nu diwawancara. 

Dina kagiatan wawancara, aya dua pihak, nyaéta \textbf{Nu Ngawawancara} (Pewawancara) jeung \textbf{Narasumber} (Informan). Dina widang jurnalistik, wawancara jadi salah sahiji cara néangan informasi keur bahan warta anu biasana dilakukeun ku wartawan.

\subsection{Klasifikasi Wawancara}
Ditilik tina jihat prak-prakanana, wawancara dibagi tilu golongan:
\begin{enumerate}
    \item \textbf{Wawancara Bébas:} Nyaéta \textbf{kagiatan wawancara anu pananya teu dirumuskeun} sacara tinulis sateuacanna. Nu ngawawancara bisa bébas nanyakeun naon baé ka nu diwawancara, tapi tetep kudu saluyu tur ati-ati sangkan teu mingpar tina tujuan awal.
    \item \textbf{Wawancara Kauger:} Anu ngawawancara téh geus dibahanan daptar pertanyaan nu lengkep katut wincikanana anu jéntré sateuacan wawancara lumangsung.
    \item \textbf{Wawancara Bébas Kauger:} Gabungan tina duanana.
\end{enumerate}

\subsection{Lengkah-Lengkah jeung Pakakas Wawancara (5W + 1H)}
Pikeun meunangkeun informasi anu jéntré, lengkah-lengkah wawancara kudu ditataan kalayan bener:
\begin{enumerate}
    \item Tahap tatan-tatan (Nangtukeun topik atawa jejer wawancara).
    \item Nangtukeun tujuan wawancara.
    \item Nangtukeun narasumber.
    \item Nangtukeun pertanyaan.
    \item Nyusun daptar pertanyaan dumasar rumus \textbf{5W + 1H}:
    \begin{itemize}
        \item \textbf{What (Naon):} Pikeun nanyakeun kajadian/barang.
        \item \textbf{Why (Kunaon):} Pikeun nanyakeun alesan.
        \item \textbf{Who (Saha):} Pikeun nanyakeun jalma/palaku.
        \item \textbf{Where (Dimana):} Pikeun nanyakeun tempat.
        \item \textbf{When (Iraha):} Pikeun nanyakeun waktu.
        \item \textbf{How (Kumaha):} Pikeun nanyakeun kaayaan/proses.
    \end{itemize}
    \item Nyiapkeun pakakas wawancara (alat tulis, alat rekaman, jeung persiapan séjén).
    \item Nyieun jangji jeung narasumber.
\end{enumerate}

\subsection{Sikep Nu Kudu Dipiboga Ku Nu Ngawawancara}
Hasil henteuna wawancara gumantung kana sikep nu ngawawancarana. Ku sikep anu hadé, wawancara bisa lumangsung akrab jeung komunikatif. Sikepna ngawengku:
\begin{itemize}
    \item \textbf{Nétral:} Teu kudu ngomentaran satuju atawa teu satuju, sabab pancénna ukur ngarekam katerangan.
    \item \textbf{Ramah:} Kudu bisa nyiptakeun suasana anu matak kataji.
    \item \textbf{Adil:} Kudu adil ka sakabéh narasumber, ulah pilih kasih upama nu diwawancara leuwih ti saurang.
    \item \textbf{Nyingkahan karengkeng:} Satékah polah nyingkahan suasana kaku/rengkeng, ulah nepikeun nu diwawancara ngarasa didakwa atawa diuji.
\end{itemize}

\section{HISTORIOGRAFI SASTRA: CARITA BABAD}
% ==============================================================

\subsection{Hakékat jeung Ciri-Ciri Utama Carita Babad}
Dina wangun prosa heubeul (buhun), \textbf{carita atawa dongeng anu ngandung unsur sajarah lokal disebut babad}. Kalayan jéntré, \textbf{carita babad téh kaasup kana wangun carita buhun anu eusina ngeunaan kajadian atawa lalakon di jaman baheula}. 

Nu jadi \textbf{ciri khas tina carita babad nyaéta miboga unsur sajarah}. Sanajan kitu, \textbf{dina carita babad, biasana aya unsur nu dicampur antara sajarah jeung pamohalan} (hal-hal anu mustahil atawa fiktif).
Kasimpulanana, \textbf{ciri ciri carita babad (lalakon baheula, pamohalan, miboga unsur sajarah)}. Dina fungsi sosialna, \textbf{tujuan carita babad nyaéta nyalamat sajarah baheula}.

\subsection{Struktur jeung Galur Carita Babad}
Struktur carita babad mibanda struktur anu sarua jeung prosa fiksi, nyaéta ngawengku:
\begin{enumerate}
    \item \textbf{Téma:} Inti pikiran atawa puseur implengan pangarang.
    \item \textbf{Amanat:} Pesen anu hayang ditepikeun ku pangarang ka nu maca.
    \item \textbf{Palaku / Tokoh:} Ngaran-ngaran boh jalma boh sasatoan nu ngalalakon.
    \item \textbf{Latar:} Nyokot kana waktu jeung tempat lumangsungna carita.
    \item \textbf{Puseur sawangan (\textit{point of view}):} Cara pangarang nyaritakeun palaku dina caritana.
    \item \textbf{Galur (Alur):} \textbf{Runtuyan pirang pirang kajadian nu ngawangun dina carita babad disebut galur}.
\end{enumerate}

Galur carita babad umumna ngagambarkeun runtuyan ieu: 
Tokoh indit lalampahan (néangan jodo/élmu) $\rightarrow$ Meunang cocoba/halangan $\rightarrow$ Meunang pitulung $\rightarrow$ Hasil ngahontal pamasudan $\rightarrow$ Mulang deui ka nagara pinanggih kabagjaan.

\subsection{Karajaan Munggaran jeung Conto Carita Babad}
Dina sajarah, \textbf{karajaan munggaran di jawa barat nyaéta tarumanagara}. 
Salian ti éta, réa conto carita babad séjénna kayaning: Babad Kawung Budug, Babad Sangkuriang, Babad Limbangan, Babad Cikundul, Babad Godog, Babad Timbanganten, jeung Babad Sumedang. Carios Prabu Siliwangi kaasup kana babad wangun prosa, sedengkeun aya ogé babad wangun pupuh saperti Wawacan Babad Timbanganten.

\section{SISTEM KEKERABATAN KULAWARGA: PANCAKAKI}
% ==============================================================

\subsection{Definisi jeung Istilah Pancakaki}
Dina budaya Sunda, \textbf{naon anu dimaksud pancakaki téh (silsilah kaluwarga)} atawa tatanan (kekerabatan) anu nuduhkeun perenahna duduluran ti hiji jalma ka jalma lianna. Istilah-istilah dasarna di antarana:
\begin{itemize}
    \item \textbf{Dulur Pet Ku Hinis:} \textbf{Istilah dulur pet kuhinis dina pancakaki hartina dulur kandung} (dulur teges saindung sabapa).
    \item \textbf{Baraya:} Sakur anu aya pancakakina.
    \item \textbf{Karuhun:} Luluhur jalma nu geus aya lila heulaeun urang (nu ngarundaykeun urang).
    \item \textbf{Bau-Bau Sinduk:} \textbf{Istilah bau bau sinduk dina pancakaki hartina baraya kénéh sanajan tos laér} (jauh).
    \item \textbf{Alo jeung Suan:} \textbf{Kaanak lanceuk disebut alo, ari ka anak adi disebut suan}.
\end{itemize}

\subsection{Rundayan Pancakaki ka Luhur jeung ka Handap}
\textbf{Rundayan ka Luhur (Ascending):} \\
Kuring $\rightarrow$ Indung/Bapa $\rightarrow$ Aki/Nini (Uyut) $\rightarrow$ \textbf{Bao} (\textbf{Kolotna buyut sok disebut bao}) $\rightarrow$ Janggawareng $\rightarrow$ Udeg-udeg $\rightarrow$ \textbf{Gantung Siwur} (\textbf{Gantung siwur dina pancakaki nyaéta bapana udeg udeg}) $\rightarrow$ Karuhun.

\textbf{Rundayan ka Handap (Descending):} \\
Kuring $\rightarrow$ Anak $\rightarrow$ Incu $\rightarrow$ Buyut $\rightarrow$ Bao $\rightarrow$ Janggawareng $\rightarrow$ Udeg-udeg.

\subsection{Istilah Hubungan Duduluran Séjénna}
\begin{itemize}
    \item \textbf{Adi beuteung:} Adina pamajikan atawa salaki.
    \item \textbf{Dulur sabrayna:} Dulur misan (anak paman / bibi).
    \item \textbf{Indung téré / Bapa téré:} Pamajikan bapa / salaki indung nu lain ngalahirkeun/ngalantarankeun urang. Dulur patétéréan nyaéta anak indung/bapa téré.
    \item \textbf{Cikal \& Bungsu:} Cikal nyaéta anak panggedéna, sedengkeun bungsu anak pangleutikna. \textit{Pangais bungsu} nyaéta lanceukna nu bungsu.
    \item \textbf{Hubungan ka gigir:} Ka lanceuk indung/bapa nyebutna \textbf{Uwa/Ua}. Ka adi indung/bapa nu awéwé nyebut \textbf{Bibi}, ari ka nu lalaki nyebut \textbf{Paman/Emang}.
    \item \textbf{Kapilanceuk \& Kapiadi:} Ka anak uwa nyebut kapilanceuk, ari ka anak bibi/emang nyebut kapiadi.
    \item \textbf{Teu hir walahir:} Teu baraya sapisan atawa deungeun-deungeun.
\end{itemize}

\section{ANALISIS PUISI MODERN: SAJAK}
% ==============================================================

\subsection{Definisi jeung Kamekaran Sajak}
Sajak téh karangan dina wangun basa ugeran. \textbf{Naon anu dimaksud sajak téh (puisi)}. Ari dina génréna mah sarua jeung pupuh atawa mantra, ngan bédana sajak mah \textbf{teu kauger ku guru lagu jeung guru wilangan}. \textbf{Saha ngaran panyajak anu kasohor, jeung dianggap salasahiji seniman sajak sastra sunda nu nyebarkeun sajak (kis ws)} atawa ngaran lengkepna Kiswanda Sastradinata.

\subsection{Struktur Lahir jeung Batin Sajak}
Sanajan teu kauger ku aturan tradisional, sajak tetep kauger ku sababaraha hal sangkan éndah:
\begin{itemize}
    \item \textbf{Padalisan:} Dina wangun fisik, \textbf{naon ari anu disebut padalisan téh (bait sajak)} atawa jajaran kalimat anu ngawangun sajak.
    \item \textbf{Diksi (Pilihan Kecap):} \textbf{Gaya bahasa anu sok dipaké dina sajak biasana bahasa éndah}. Loba kekecapan harti injeuman anu mangrupa perlambang (simbol).
    \item \textbf{Wirahma jeung Purwakanti:} Pilihan kecap anu bisa nyawangkeun sora wirahma nepi ka genah dibacanana.
    \item \textbf{Imaji:} \textbf{Naon ari anu disebut imaji téh (citraan atawa gambaran angen-angen/pikiran)} nu maca.
    \item \textbf{Amanat:} \textbf{Nu disebut amanan dina sajak téh hartina pesan moral} anu hayang ditepikeun panyajak.
\end{itemize}

\subsection{Tujuan jeung Langkah Nulis Sajak}
Pikeun kaperluan ékspresi, \textbf{sebutkeun salasahiji tujuan nyieun sajak jang nyatakeun rasa, pikiran jeung pangalaman}. \\

Pikeun nyieun sajak anu alus, \textbf{saméméh rék nulis sajak aya sabaraha hal anu kudu dicangkum (6)} perkara:
\begin{enumerate}
    \item Nangtukeun téma.
    \item Nangtukeun suasana atawa milarian inspirasi.
    \item Milih diksi (pilihan kecap anu éndah).
    \item Ngolah imaji.
    \item Ngatur wirahma.
    \item Nangtukeun amanat.
\end{enumerate}
