    % 1. Eksponen
    \item Bentuk sederhana dari $\displaystyle \frac{(a^2 b^3)^2}{a^3 b^5}$ adalah \dots
    \begin{multicols}{2}
    \begin{enumerate}[label=\Alph*.]
        \item $a b$
        \item $a^2 b$
        \item $a b^2$
        \item $a^2 b^2$
        \item $a^{-1} b$
    \end{enumerate}
    \end{multicols}

    % 2. Logaritma
    \item Nilai dari ${}^2\log 8 + {}^3\log 9 - {}^5\log 1$ adalah \dots
    \begin{multicols}{2}
    \begin{enumerate}[label=\Alph*.]
        \item $2$
        \item $3$
        \item $4$
        \item $5$
        \item $6$
    \end{enumerate}
    \end{multicols}

    % 3. SPLDV
    \item Diketahui sistem persamaan linear dua variabel $2x + y = 7$ dan $x - y = 2$. Nilai dari $x + 2y$ adalah \dots
    \begin{multicols}{2}
    \begin{enumerate}[label=\Alph*.]
        \item $3$
        \item $4$
        \item $5$
        \item $6$
        \item $7$
    \end{enumerate}
    \end{multicols}

    % 4. Persamaan linear 1 variabel
    \item Penyelesaian dari persamaan linear $3(2x - 1) = 4x + 5$ adalah \dots
    \begin{multicols}{2}
    \begin{enumerate}[label=\Alph*.]
        \item $x = 1$
        \item $x = 2$
        \item $x = 3$
        \item $x = 4$
        \item $x = 5$
    \end{enumerate}
    \end{multicols}

    % 5. SPLTV
    \item Diberikan sistem persamaan linear tiga variabel berikut:
    \begin{align*}
        x + y + z &= 6 \\
        x - y + z &= 2 \\
        x + y - z &= 0
    \end{align*}
    Nilai dari $x \cdot y \cdot z$ adalah \dots
    \begin{multicols}{2}
    \begin{enumerate}[label=\Alph*.]
        \item $3$
        \item $6$
        \item $8$
        \item $10$
        \item $12$
    \end{enumerate}
    \end{multicols}

    % 6. Bentuk umum fungsi kuadrat
    \item Di antara bentuk persamaan berikut, yang merupakan bentuk umum fungsi kuadrat adalah \dots
    \begin{enumerate}[label=\Alph*.]
        \item $f(x) = ax + b$
        \item $f(x) = ax^2 + bx + c$, dengan $a \neq 0$
        \item $f(x) = a^x$
        \item $f(x) = \log(x)$
        \item $f(x) = \frac{a}{x} + b$
    \end{enumerate}

    % 7. Sumbu simetri
    \item Sumbu simetri dari grafik fungsi kuadrat $f(x) = x^2 - 6x + 8$ adalah \dots
    \begin{multicols}{2}
    \begin{enumerate}[label=\Alph*.]
        \item $x = -6$
        \item $x = -3$
        \item $x = 2$
        \item $x = 3$
        \item $x = 6$
    \end{enumerate}
    \end{multicols}

    % 8. Titik puncak dan arah parabola (Divisualisasikan secara estetis)

        \item $(2, -1)$

    \item Perbedaan utama antara histogram dan diagram batang biasa terletak pada \dots

        \item Tidak ada perbedaan antara keduanya
    \end{enumerate}
