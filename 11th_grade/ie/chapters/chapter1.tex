\section{Elementary 2 English}
\subsection{Passive Voice}
\begin{definition}[Passive Voice]
  The subject of the sentence recieves the action of a verb, rather than doing the action.
\end{definition}
\begin{enumerate}
\item Present tense:
  $$\text{ Subject } + \text{ to be (are/am/is) } + \text{ verb 3 } $$
\item Past tense:
  $$\text{ Subject } + \text{ was/were } + \text{ verb 3 }$$
\end{enumerate}

\subsection{Abstract Noun}
\begin{definition}[Abstract Noun]
  Is a noun to be made from adjective, verb, or the noun itself.
\end{definition}
\begin{enumerate}
\item Adjective + (ness, y, ity, al, ion, ence, ance, dom, ment) $\to$ noun
\item Verb(ing, er/or) $\to$ noun
\item Noun + (dom, al, ion, hood, ness, ence, ance, ship) $\to$ noun
\end{enumerate}
\textbf{Note: } After the words of
\begin{enumerate}
\item Article (a, an, the)
\item To have (have, has, had)
\item Certain verbs (enjoy, live, finish, start)
\item Apostrophe ('s, of)
\item Ordinal number $(1^{st}, \, 2^{nd}, \, 3^{rd})$
\item Cardinal number (1, 2, 3)
\item Possesive adjective (my, your, our, their, his, its)
\item Adjective of quantity (er, est, more, the most)
\item Conjunction (for, because, because of)
\item Proposition (in on, at, above, under)
\end{enumerate}
$$\downarrow$$
\begin{center}
  Noun
\end{center}

\subsection{Apostrophe}
\begin{definition}[Apostrophe]
  Possestion (ownership), to create contractions (missing letters), or to form certain plurals.
\end{definition}
  $A = \text{ Alive}$ \\
  $N = \text{ Not alive}$ \\
\begin{enumerate}
\item $A \land A = \text{ 's}$
\item $A \land N = \text{ 's}$
\item $N \land A = \text{ of}$
\item $N \land M = \text{ of}$
\end{enumerate}

\subsection{Epithet Adjective}
\begin{definition}[Ephitet Adjective]
  A descriptive word or phrase added to a person, place or thing to highlight a key characteristic or quality.
\end{definition}
$$\text{Adj } + \text{ Noun}$$
\textbf{Note: }
\begin{enumerate}
\item The processing of adjective:
  \begin{itemize}
  \item Real adjective $\to$ $V + \text{ to be}$
  \item $V + \text{ ing } \to $ past participle
  \end{itemize}
\item The processing of unit:
  \begin{itemize}
  \item Real noun
  \item Abstract + noun
  \end{itemize}
\item Noun in epithet adjective are not allowed to be taken from a personal pronoun.
  \begin{table}[H]
    \centering
    \begin{tabular}{cc}
      \toprule
      \textbf{Personal Pronoun} & \textbf{Balancing} \\ \midrule
      I & Girl, Boy, Man, Woman \\
      You, We, They & People, Citizen \\
      He & Boy, Man \\
      It & Animal, Thing \\ \bottomrule
    \end{tabular}
  \end{table}
\end{enumerate}
