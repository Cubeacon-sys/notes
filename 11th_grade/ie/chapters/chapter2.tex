\subsection{Supposed To}
\begin{definition}[Supposed To]
  Expected, required, or planned to do something.
\end{definition}
\begin{enumerate}
\item Present:
  $$S + \text{ supposed to } + ...$$
\item Past:
  $$S + \text{ supposed to } + ...$$
\item Passive:
  $$
  S + \text{ to be } + \text{ supposed to } + ...
  \begin{cases}
    \text{by + o} \\
    \text{to inf} \\
    \text{adv}
  \end{cases}
  $$
\end{enumerate}

\subsection{Used To}
\begin{definition}[Used To]
  Describes past habits, repeated actions, or states that were true in the past but are no longer true or relevant in the present.
\end{definition}
\begin{enumerate}
\item Used to (Back then):
  $$
  S + \text{ used to } +
  \begin{cases}
    \text{bare inf} \\
    \text{be } + \text{ out of verb}
  \end{cases}
  $$
\item Used to (Habit):
  $$
  S + \text{ to be } + \text{ used to } +
  \begin{cases}
    \text{verb -ing} \\
    \text{being } + \text{ out of verb}
  \end{cases}
  $$
\end{enumerate}

\subsection{Suppose to}
\begin{definition}[Suppose to]
  Expectations that if not done, will be dissapointing.
\end{definition}
\begin{enumerate}
\item Present:
  $$
  S + \text{ suppose } + o +
  \begin{cases}
    \text{to inf} \\
    \text{to be } + \text{ out of verb}
  \end{cases}
  $$
\item Past tense:
  $$
  S + \text{ supposed } + o +
  \begin{cases}
    \text{to inf} \\
    \text{to be } + \text{ out of verb}
  \end{cases}
  $$
\end{enumerate}

\subsection{Exclamatory sentence}
\begin{definition}[Exclamatory sentence]
  Expresses strong emotion. Always ends with an exclamation mark (!).
\end{definition}
\begin{enumerate}
\item
  $$\text{How } + \text{ adjective } + S \text{ to be } + ... + ! $$
\item
  $$\text{What } + \text{ a/an/the } + \text{ epithet adjective } + .. + ! $$
\end{enumerate}
