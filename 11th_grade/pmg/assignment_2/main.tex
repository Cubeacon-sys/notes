  using namespace std;

  int main() {
    string nama, alamat;
    int usia, tinggi, berat;
    
    cout << "Masukan Nama: ";
    cin >> nama;
    cout << "Masukan Usia: ";
    cin >> usia;
    cout << "Masukan Alamat: ";
    cin >> alamat;
    cout << "Masukan Tinggi: ";
    cin >> tinggi;
    cout << "Masukan Berat Badan: ";
    cin >> berat;
    
    cout << "Nama anda adalah: " << nama << endl
    << "Usia anda adalah: " << usia << endl
    << "Alamat anda adalah: " << alamat << endl
    << "Tinggi anda adalah: " << tinggi << endl
    << "Berat badan anda adalah: " << berat << endl;
    return 0;
  }
\end{lstlisting}
