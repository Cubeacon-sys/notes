\subsection{Pembagian Fungsi}
\begin{definition}[Pembagian Fungsi]
  $$\frac{f}{g}(x) = \frac{f(x)}{g(x)}$$
  Dengan syarat $g(x) \ne 0$.
\end{definition}

Ada 3 langkah mengerjakan fungsi aljabar yaitu:
\begin{enumerate}
\item Menuliskan nilai $\frac{f(x)}{g(x)}$.
\item Menyederhanakan pecahan (memfaktorkan).
\item Membagi dan menuliskan hasil akhir.
\end{enumerate}

\subsubsection{Cara Memfaktorkan}
\begin{enumerate}
\item Ambil angka/variabel yang sama dari semua suku. \\
  \begin{example}[Variabel yang Sama]
    \hfill
    \begin{enumerate}
    \item $6x + 9 = 3(2x + 3)$
    \item $4x^2 - 8x = 4x(x - 2x)$
    \item $x^3 + 2x^2 - x^2(x + 2)$
    \end{enumerate}
  \end{example}
\item Selisih 2 kuadrat. \\
  $$a^2 - b^2 = (a + b) (a - b)$$
  \begin{example}[Selisih 2 Kuadrat]
    \hfill
    \begin{enumerate}
    \item $x^2 - 9 = (x + 3) (x - 3)$
    \item $x^2 - 14 = (x + 4) (x - 4)$
    \item $9x^2 - 25 = (3x + 5) (3x - 5)$
    \end{enumerate}
  \end{example}
\item Trinomial \\
  $$ax^2 + bx + c = 0 \quad (a = 1)$$
  Carikan 2 bilangan yang dikalikan hasilnya $c$ dan dijumlahkan hasilnya $b$.
  \begin{example}[Trinomial]
    \hfill \\
    $x^2 5x = 0$ \\
    $6 \cdot 1 = 6 \quad 6 + 1 = 7$ \\
    $3 \cdot 2 = 6 \quad 3 + 2 = 5$ \\
    $(x + 3) (x + 2)$
  \end{example}
\item Trinomial $a \ne 1$
  \begin{enumerate}
  \item Pertama hitung $a \cdot c$.
  \item Cari 2 bilangan yag dikalikan hasilnya $ac$.
  \item Pecahkan suku tengah menjadi 2 lalu kelompokkan.
  \end{enumerate}
  \begin{example}[Trinomial $a \ne 1$]
    \hfill \\
    $\left( \frac{f}{g}\right)(x) = \frac{6x^2 + x - 2}{2x - 1}$ \\
    $\frac{6x^2 - 3x + 4x - 2}{2x - 1}$ \\
    $\frac{3x(2x - 1) + 2(2x - 1)}{2x - 1}$ \\
    $\frac{(3x + 2) (2x - 1)}{2x - 1}$ \\
    $3x + 2$
  \end{example}
\end{enumerate}
