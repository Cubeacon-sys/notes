\subsection{Komposisi Fungsi}
\begin{definition}[Komposisi Fungsi]
  Komposisi dari fungsi $f$ dan $g$ didefinisikan sebagai berikut:
  \[(f \circ g)(x) = f(g(x))\]
  \[(g \circ f)(x) = g(f(x))\]
  \textit{(Menggabungkan 2 fungsi, memasukkan fungsi pertama ke fungsi ke dua)}
\end{definition}

\begin{example}[Komposisi Fungsi]
  \[f(x) = x^2 + 3x - 2\]
  \[g(x) = x - 1\]
  \begin{enumerate}
  \item $f \circ g(x) = (x - 1)^2 + 3(x - 1) - 2$ \\
    $(x - 1) (x - 1) + 3x - 3 - 2$ \\
    $(x^2 - x - x + 1) + 3x - 5$ \\
    $x^2 - 2x + 1 + 3x - 5$ \\
    $x^2 + x - 4$ 
  \item $g \circ f(x) = (x^2 + 3x - 2) - 1$ \\
    $x^2 + 3x - 2 - 1$ \\
    $x^2 + 3x - 3$
  \end{enumerate}
\end{example}

\subsection{Komposisi dari 3 Fungsi}
\begin{definition}[Komposisi dari 3 Fungsi]
  \[(f \circ g) \circ h = f \circ (g \circ h)\]
\end{definition}

\begin{example}[Komposisi dari 3 Fungsi]
  \[f(x) = x - 1\]
  \[g(x) = x^2 - 3x - 2\]
  \[h(x) = x + 5\]
  $(f \circ g)(x) = f(g(x))$ \\
  $(x^2 + 3x - 2) - 1$ \\
  $x + 3x - 2 - 1$ \\
  $x + 3x - 3$ \\
  $(f \circ g) \circ h(x) = f \circ g(h(x))$ \\
  $(x + 5)^2 + 3(x + 5) - 3$ \\
  $(x + 5) (x + 5) + 3 (x + 5) - 3$ \\
  $x^2 + 5x + 5x + 25 + 3x + 15 - 3$ \\
  $x^2 + 13x + 37$ \\
\end{example}
