\subsection{Pengurangan Fungsi}
\begin{definition}[Pengurangan Fungsi]
  $$(f - g)(x) = f(x) - g(x)$$
\end{definition}
\begin{example}[Pengurangan Fungsi]
  \hfill
  \begin{enumerate}
  \item $f(x) = 5x + 7$ \\
    $g(x) = 3x - 2$ \\
    $(f - g)(x) = (5x + 7) - (3x - 2)$ \\
    $5x + 7 - 3x - 2$ \\
    $5z - 3x + 7 + 2$ \\
    $2x + 9$
  \item $f(x) = 2x^2 - 6x + 10$ \\
    $g(x) = x^2 + 4x - 5$ \\
    $(f - g)(x) = (2x^2 - 6x + 10) - (x^2 + 4x - 5)$ \\
    $2x^2 - 6x + 10 - x^2 + 4x - 5$ \\
    $2x^2 - x^2 - 6x - 4x + 10 + 5$ \\
    $x^2 - 10x + 15$
  \end{enumerate}
\end{example}

\subsection{Perkalian Fungsi}
\begin{definition}[Perkalian Fungsi]
  $$ax(b + c) = ab + ac$$
\end{definition}
\begin{example}[Perkalian Fungsi]
  \hfill \\
  $f(x) = 2x$ \\
  $g(x) = x - 7$ \\
  $(f \cdot g)(x) = 2x \cdot (x - 7)$ \\
  $2x^2 - 14x$ \\
  $(g \cdot f)(x) = (x - 7) \cdot 3x$ \\
  $2x^2 - 14x$ \\
\end{example}
\begin{definition}[Perkalian Fungsi]
  $$(a + b) \cdot (a + b) = a^2 + ab + ba + b^2 \to a^2 + 2ab + b^2$$
\end{definition}
\begin{example}[Perkalian Fungsi]
  \hfill \\
  $f(x) = 2x - 3$ \\
  $g(x) = 3x + 4$ \\
  $(f \cdot g)(x) = (2x - 3) \cdot (3x + 4)$ \\
  $6x^2 + 8x - 9x - 12$ \\
  $6x^2 - x - 12$ 
\end{example}
\textbf{Note: }Aturan saat mengalikan:
\begin{enumerate}
\item Gunakan sifat distributif.
\item Jika variabelnya sama. Pangkat dijumlahkan $(x^m + x^n = x^{m + n})$.
\item Kalikan angka dengan angka.
\item Kelompokan suku sejenis.
\item Sederhanakan.
\end{enumerate}
