\begin{enumerate}
    \item Notasi yang menunjukkan nilai fungsi untuk $x=8$ adalah \dots
    \begin{enumerate}
        \item $f(8)$
        \item $f(x+8)$
        \item $8f(x)$
        \item $f(8x)$
        \item $f(x)-8$
    \end{enumerate}

    \item Jika diketahui fungsi $f(x)=2x+7$. Berapa nilai $f(9)$ adalah\dots
    \begin{enumerate}
        \item 18
        \item 20
        \item 22
        \item 25
        \item 35
    \end{enumerate}

    \item Jika diketahui fungsi $f(x)=x^2-2x+1$. Tentukan nilai $f(5)$ adalah \dots
    \begin{enumerate}
        \item 1
        \item 6
        \item 10
        \item 16
        \item 25
    \end{enumerate}

    \item Diketahui fungsi $f(x)$ dan $g(x)$. Bentuk yang tepat untuk menentukan hasil pengurangan kedua fungsi adalah\dots
    \begin{enumerate}
        \item $(f-g)(x)=f(x)+g(x)$
        \item $(f-g)(x)=f(x)-g(x)$
        \item $(f-g)(x)=f(x)\times g(x)$
        \item $(f-g)(x)=\frac{f(x)}{g(x)}$
        \item $(f-g)(x)=f(g(x))$
    \end{enumerate}

    \item Diketahui $f(x)=5x+5$ dan $g(x)=x-1$. Bentuk dari $(f+g)(x)$ adalah \dots
    \begin{enumerate}
        \item $6x-4$
        \item $6x+4$
        \item $6x+6$
        \item $6x-6$
        \item $4x-4$
    \end{enumerate}

    \item Diketahui $f(x)=x^2+x$ dan $g(x)=2x-1$. Nilai dari $(f+g)(2)$ adalah \dots
    \begin{enumerate}
        \item 6
        \item 9
        \item 8
        \item $-9$
        \item 10
    \end{enumerate}

    \item Diketahui $f(x)=x+2$ dan $g(x)=2x-4$. Bentuk sederhana dari $(f-g)(x)$ adalah\dots
    \begin{enumerate}
        \item $3x+6$
        \item $-3x-6$
        \item $-x+6$
        \item $-x-6$
        \item $x-6$
    \end{enumerate}

    \item Diketahui $f(x)=x^2+2x+2$ dan $g(x)=x+3$. Nilai dari $(f-g)(-1)$ adalah \dots
    \begin{enumerate}
        \item $-1$
        \item 1
        \item 5
        \item 7
        \item $-7$
    \end{enumerate}

    \item Diketahui $f(x)=3x^2-4x+5$ dan $g(x)=x^2+2x-3$. Dua siswa menghitung $f(x)-g(x)$.\\
    Siswa A memperoleh: $2x^2-6x+8$\\
    Sedangkan Siswa B memperoleh: $2x^2-2x+2$\\
    Analisis hasil pekerjaan kedua siswa tersebut. Hasil yang benar adalah\dots
    \begin{enumerate}
        \item Siswa A, karena semua suku pada $g(x)$ dikurangkan
        \item Siswa B, karena koefisien $x$ dikurangkan
        \item Siswa A, karena $3x^2-x^2=2x^2$ dan $-4x-2x=-6x$
        \item Siswa B, karena tanda negatif tidak memengaruhi konstanta
        \item Keduanya benar
    \end{enumerate}

    \item Diketahui $f(x)=x+3$ dan $g(x)=x-5$. Bentuk sederhana dari $(fg)(x)$ adalah \dots
    \begin{enumerate}
        \item $x^2-2x+15$
        \item $x^2-2x-15$
        \item $x^2+2x-15$
        \item $x^2+2x+15$
        \item $-x^2-2x+15$
    \end{enumerate}

    \item Diketahui $f(x)=x+1$ dan $g(x)=2x-1$. Nilai dari $(fg)(3)$ adalah \dots
    \begin{enumerate}
        \item 10
        \item 9
        \item 12
        \item 15
        \item 20
    \end{enumerate}

    \item Uraian dari bentuk $(2x+1)^2(x-3)$. Bentuk yang benar setelah diuraikan dan disederhanakan adalah\dots
    \begin{enumerate}
        \item $4x^3-8x^2-11x-3$
        \item $4x^3-12x^2-8x-35$
        \item $4x^3-8x^2-8x-3$
        \item $4x^3+8x^2-11x+3$
        \item $2x^3-8x^2-11x-3$
    \end{enumerate}

    \item Diketahui fungsi $f(x)$ dan $g(x)$ dengan $g(x)\neq 0$. Bentuk operasi pembagian kedua fungsi tersebut adalah \dots
    \begin{enumerate}
        \item $(f+g)(x)=f(x)+g(x)$
        \item $(f-g)(x)=f(x)-g(x)$
        \item $(fg)(x)=f(x)\times g(x)$
        \item $\left(\frac{f}{g}\right)(x)=\frac{f(x)}{g(x)}$
        \item $(f\circ g)(x)=f(g(x))$
    \end{enumerate}

    \item Diketahui $f(x)=x^2-7x+10$ dan $g(x)=x-2$. Hasil dari $\left(\frac{f}{g}\right)(x)$ adalah\dots
    \begin{enumerate}
        \item $x+5$
        \item $-x+5$
        \item $x-5$
        \item $-x-5$
        \item $x-2$
    \end{enumerate}

    \item Diketahui $f(x)=2x^2+5x-3$ dan $g(x)=x+3$. Bentuk $\left(\frac{f}{g}\right)(5)$ adalah \dots
    \begin{enumerate}
        \item 9
        \item 10
        \item 20
        \item 25
        \item 30
    \end{enumerate}

    \item Diketahui fungsi $f$ dan $g$. Notasi yang tepat untuk menyatakan komposisi fungsi $f$ terhadap $g$ adalah\dots
    \begin{enumerate}
        \item $(f+g)(x)$
        \item $(f-g)(x)$
        \item $(f\circ g)(x)$
        \item $(g\circ f)(x)$
        \item $(f\times g)(x)$
    \end{enumerate}

    \item Jika fungsi $f: \mathbb{R}\to\mathbb{R}$ dan $g: \mathbb{R}\to\mathbb{R}$ ditentukan oleh $f(x)=x-2$ dan $g(x)=x^2+x+16$, maka $(g\circ f)(x)=\dots$
    \begin{enumerate}
        \item $x^2+3x-18$
        \item $x^2+3x+18$
        \item $x^2-3x+18$
        \item $x^2-3x-18$
        \item $x^2+3x+16$
    \end{enumerate}

    \item Jika fungsi $f: \mathbb{R}\to\mathbb{R}$ dan $g: \mathbb{R}\to\mathbb{R}$ ditentukan oleh $f(x)=3x^2-x+2$ dan $g(x)=2x-3$, maka $(f\circ g)(x)=$
    \begin{enumerate}
        \item $12x^2-38x+32$
        \item $12x^2-38x-32$
        \item $12x^2+38x+32$
        \item $13x^2-38x+32$
        \item $13x^2+38x+32$
    \end{enumerate}

    \item Diketahui $f(x)=x^2-2$ dan $g(x)=2x+1$. Jika $(f\circ g)(x)=7$, nilai $x$ yang memenuhi adalah\dots
    \begin{enumerate}
        \item $x = 1$ atau $x = -2$
        \item $x = 1$ atau $x = -1$
        \item $x = 2$ atau $x = 1$
        \item $x = 3$ saja
        \item $x = -1$ saja
    \end{enumerate}

    \item Diketahui $f(x)=2x+3$, $g(x)=x^2-1$, dan $h(x)=x-2$. Hasil dari $(f\circ g\circ h)(x)$ yang benar adalah \dots
    \begin{enumerate}
        \item $2x^2-8x+9$
        \item $2x^2-8x+7$
        \item $2x^2-8x+5$
        \item $x^2-4x+5$
        \item $2x^2-4x+5$
    \end{enumerate}

    \item Diketahui $f(x)=3x-2$, $g(x)=x^2+x-1$, dan $h(x)=x+1$. Hasil dari $(g\circ f\circ h)(x)$ adalah \dots
    \begin{enumerate}
        \item $9x^2+9x+1$
        \item $9x^2+9x-2$
        \item $9x^2+3x-2$
        \item $3x^2+9x-2$
        \item $9x^2-9x-2$
    \end{enumerate}

    \item Diketahui $f(x)=3x-2$ dan $(f\circ g)(x)=6x+7$. Fungsi $g(x)$ yang sesuai adalah \dots
    \begin{enumerate}
        \item $g(x)=2x+1$
        \item $g(x)=2x+2$
        \item $g(x)=2x+3$
        \item $g(x)=2x+4$
        \item $g(x)=2x+5$
    \end{enumerate}

    \item Diketahui $g(x)=x-2$ dan $(f\circ g)(x)=x^2-4x+7$. Fungsi $f(x)$ yang benar adalah \dots
    \begin{enumerate}
        \item $f(x)=x^2+3$
        \item $f(x)=x^2-3$
        \item $f(x)=x^2+2$
        \item $f(x)=x^2-4x+7$
        \item $f(x)=x^2-2$
    \end{enumerate}

    \item Diketahui $f(x)=x-4$ dan $(f\circ g)(x)=x^2-2x+3$. Rumus $g(x)$ yang tepat adalah \dots
    \begin{enumerate}
        \item $g(x)=x^2-2x+7$
        \item $g(x)=x^2+3x+7$
        \item $g(x)=-x^2-3x+7$
        \item $g(x)=x^2+2x+7$
        \item $g(x)=x^2-3x+7$
    \end{enumerate}

    \item Diketahui suatu fungsi $f$ memiliki fungsi invers. Notasi yang tepat untuk menyatakan fungsi invers dari $f(x)$ adalah \dots
    \begin{enumerate}
        \item $f(x)$
        \item $f^{-1}(x)$
        \item $f^2(x)$
        \item $\frac{1}{f(x)}$
        \item $f(x^{-1})$
    \end{enumerate}

    \item Diketahui fungsi $f(x)=\frac{4x-5}{2x+3}$, $x\neq -\frac{3}{2}$. Fungsi invers $f^{-1}(x)$ yang benar adalah\dots
    \begin{enumerate}
        \item $f^{-1}(x)=\frac{3x+5}{4-2x}$
        \item $f^{-1}(x)=\frac{3x-5}{4+2x}$
        \item $f^{-1}(x)=\frac{4x+5}{3-2x}$
        \item $f^{-1}(x)=\frac{-3x+5}{2x+4}$
        \item $f^{-1}(x)=\frac{2x+3}{4x-5}$
    \end{enumerate}

    \item Ira memikirkan sebuah bilangan positif $x$. Bilangan tersebut dikuadratkan, ditambah 4, lalu dikalikan 2. Fungsi $f(x)$ dan $f^{-1}(x)$ yang sesuai adalah \dots
    \begin{enumerate}
        \item $f(x)=2x^2+8$ dan $f^{-1}(x)=\sqrt{\frac{x-8}{2}}$
        \item $f(x)=2x^2+4$ dan $f^{-1}(x)=\sqrt{\frac{x-4}{2}}$
        \item $f(x)=2x^2-8$ dan $f^{-1}(x)=\sqrt{\frac{x+8}{2}}$
        \item $f(x)=2(x+4)^2$ dan $f^{-1}(x)=\sqrt{\frac{x}{2}}-4$
        \item $f(x)=x^2+8$ dan $f^{-1}(x)=\sqrt{x-8}$
    \end{enumerate}

    \item Banyaknya ruang penyimpanan yang digunakan dimodelkan dengan $f(x)=3x+12$ (GB). Jika kapasitas penyimpanan yang digunakan $f(x) = 60$ GB, maka ukuran data $x$ adalah \dots
    \begin{enumerate}
        \item 16 GB
        \item 12 GB
        \item 14 GB
        \item 18 GB
        \item 20 GB
    \end{enumerate}

    \item Diketahui fungsi $f(x)=3x+4$ dan $g(x)=\frac{4x+5}{3-2x}$. Bentuk $(f\circ g)^{-1}(x)$ yang benar adalah\dots
    \begin{enumerate}
        \item $(f\circ g)^{-1}(x)=\frac{3x-27}{2x+4}$
        \item $(f\circ g)^{-1}(x)=\frac{3x-27}{4-2x}$
        \item $(f\circ g)^{-1}(x)=\frac{3x+27}{2x-4}$
        \item $(f\circ g)^{-1}(x)=\frac{4x+27}{3-2x}$
        \item $(f\circ g)^{-1}(x)=\frac{2x+4}{3x-27}$
    \end{enumerate}

    \item Jika $f(x)=3x$ dan $g(x)=5-2x$, maka nilai $(g\circ f)^{-1}(17)=\dots$
    \begin{enumerate}
        \item $-2$
        \item $-1$
        \item 0
        \item 1
        \item 2
    \end{enumerate}
\end{enumerate}
