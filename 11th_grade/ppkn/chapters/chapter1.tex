\section{Menerapkan Sila-Sila Pancasila}

\subsection{Rumusan Dasar Negara Indonesia Merdeka}

\subsubsection{Perumusan Dasar Negara}
\begin{itemize}
\item Sidang pertama BPUPK dilaksanakan pada tanggal 29 Mei hingga 1 Juni 1945 dengan agenda membahas dasar negara Indonesia merdeka. Ketua BPUPK KRT Radjiman Wedyodiningrat meminta para anggota menyampaikan pandangannya.
\item Pembentukan Panitia Kecil (Panitia Delapan) di bawah pimpinan Ir. Sukarno untuk menampung serta memeriksa usul-usul tertulis dari anggota BPUPK.
\item Pembentukan Panitia Sembilan untuk mempertemukan pandangan antara golongan kebangsaan dan golongan Islam.
\item Anggota Panitia Sembilan terdiri atas Ir. Sukarno, Drs. Mohammad Hatta, Mr. A.A. Maramis, Abikoesno Tjokrosoejoso, KH Abdul Kahar Moezakir, H. Agus Salim, Mr. Ahmad Soebardjo, KH Wachid Hasjim, dan Mr. Mohammad Yamin.
\item Panitia Sembilan menyetujui rancangan Pembukaan UUD pada tanggal 22 Juni 1945 yang dikenal dengan sebutan Piagam Jakarta (Jakarta Charter).
\end{itemize}

\subsubsection{Pengesahan Dasar Negara}
\begin{itemize}
\item Setelah kekalahan Jepang dari Sekutu, BPUPK dibubarkan dan dibentuk Panitia Persiapan Kemerdekaan Indonesia (PPKI) pada tanggal 7 Agustus 1945 dengan Ketua Ir. Sukarno dan Wakil Ketua Drs. Mohammad Hatta.
\item Setelah Proklamasi Kemerdekaan pada tanggal 17 Agustus 1945, PPKI menggelar sidang untuk menentukan dan mengesahkan konstitusi negara.
\item Pada tanggal 18 Agustus 1945, dilakukan perubahan mendasar terhadap naskah Piagam Jakarta demi menjaga persatuan dan kesatuan bangsa:
  \begin{enumerate}
  \item Kata Muqaddimah diganti dengan kata Pembukaan.
  \item Kalimat "dengan kewajiban menjalankan syariat Islam bagi pemeluk-pemeluknya" pada alinea keempat diganti menjadi "Ketuhanan Yang Maha Esa".
  \item Redaksi sila kedua disempurnakan menjadi "Kemanusiaan yang adil dan beradab".
  \item Pasal 6 ayat (1) yang semula berbunyi "Presiden ialah orang Indonesia asli dan beragama Islam" diganti menjadi "Presiden ialah orang Indonesia Asli".
  \item Pasal 29 ayat (1) disesuaikan menjadi "Negara berdasar atas Ketuhanan Yang Maha Esa".
  \end{enumerate}
\item Hasil keputusan penting Sidang PPKI (18 hingga 20 Agustus 1945):
  \begin{itemize}
  \item Sidang 18 Agustus 1945: Mengesahkan UUD NRI Tahun 1945, memilih Ir. Sukarno sebagai Presiden dan Drs. Mohammad Hatta sebagai Wakil Presiden, serta membentuk Komite Nasional.
  \item Sidang 19 Agustus 1945: Membagi wilayah Indonesia menjadi 8 provinsi, membentuk Komite Nasional Daerah, dan menetapkan 12 kementerian.
  \item Sidang 20 Agustus 1945: Membentuk Komite Nasional Pusat, Partai Nasional Indonesia, dan Badan Keamanan Rakyat.
  \end{itemize}
\end{itemize}

\subsubsection{ Nilai-Nilai Luhur dalam Perumusan dan Pengesahan Dasar Negara}
\begin{itemize}
\item Menghargai Pendapat: Anggota sidang mendengarkan pendapat dari pimpinan maupun anggota lain secara sopan, jelas, dan tertib tanpa memotong pembicaraan.
\item Rendah Hati: Para tokoh tidak egois dengan pandangan masing-masing dan bersedia menerima kesepakatan yang lebih sesuai untuk kepentingan bangsa dan negara.
\end{itemize}
