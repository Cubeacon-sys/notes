\subsection{Evaluating Functions}
\textbf{Note: }
$$y = -7x + 5 \equiv f(x) -7x + 5$$
$$y = \frac{3}{2} x - 1 \equiv g(x) = \frac{3}{2} x - 1$$
We use the $f(x)$ notation to know what the input is.

\begin{example}[Exponents]
  $$h(x) = -2x^2 + x - 1$$
  \begin{enumerate}
  \item $h(-1) = -2(-1)^2 + (-1) - 1$ \\
    $h(-1) = -4$ \\
    \textbf{$(-1, -4)$}
  \item $h(-x) = -2(-x)^2 + (-x) - 1$ \\
    $h(-x) = -2x^2 + x - 1$
  \item $h(x + 1) = -2(x + 1)^2 + (x + 1) - 1$ \\
    $h(x + 1) = -2(x^2 + 2x + 1) + (x + 1) - 1$ \\
    $h(x + 1) = -2x^2 - 4x - 2 + x + 1 - 1$ \\
    $h(x + 1) = -2x^2 - 3x - 1$
  \end{enumerate}
\end{example}

\begin{example}[Square Root]
  $$f(x) = \sqrt{x^2 - 3x}$$
  \begin{enumerate}
  \item $f(5) = \sqrt{(5)^2 - 3(5)}$ \\
    $f(5) = \sqrt{10}$ \\
    $(5, \sqrt{10})$
  \item $f(0) = \sqrt{(0)^2 - 3(0)}$ \\
    $f(0) = \sqrt{0}$ \\
    $(0, 0)$
  \item $f(2x) = \sqrt{(2x)^2 - 3(2x)}$ \\
    $f(2x) = \sqrt{4x^2 - 6x}$
  \item $f(x + h) = \sqrt{(x + h)^2 - 3(x + h)}$ \\
    $f(x + h) = \sqrt{x^2 + 2xh + h^2 - 3x - 3h}$
  \item $f(1) = \sqrt{(1)^2 - 3(1)}$ \\
    $f(1) = \sqrt{-2}$ \\
    \textbf{Note: } This is a domain issue so we can't use 1.
  \end{enumerate}
\end{example}

\textbf{Note: } For domains:


\begin{enumerate}
\item \textbf{Square Roots: } The inside of the square root \textit{(radicand)} must always always positive.
\item \textbf{Denominator: }  The denominator must never be 0.
\end{enumerate}

\begin{example}[Fractions]
  $$g(x) = \frac{2x + 1}{x - 5}$$
  \begin{enumerate}
  \item $g(0) = \frac{2(0) + 1}{(0) - 5}$ \\
    $g(0) = \frac{1}{-5}$
  \item $g(x^2) = \frac{2(x^2) + 1}{(x^2) - 5}$ \\
    $g(x^2) = \frac{2x^2 + 1}{x^2 - 5}$
  \item $g(5) = \frac{2(5) + 1}{(5) - 5}$ \\
    $g(5) = \frac{11}{0}$ \\
    \textbf{Note: } This is also a domain issue.
  \item $-g(x) = - \frac{2x + 1}{x - 5}$
  \end{enumerate}
\end{example}

\subsubsection{Difference Quotient}
\begin{definition}[Difference Quotient]
  A formula used in algebra and calculus to find the \textbf{average rate of change} of a function over a small interval
  $$\frac{f(x + h) - f(x)}{h}$$
\end{definition}

\begin{example}[Example]
  $$f(x) = x^2 + 1$$
  $f(x + h) = (x + h)^2 + 1$ \\
  $f(x + h) = x^2 + xh + xh + h^2 + 1$ \\
  $f(x + h) = x^2 + 2xh + h^2 + 1$ \\


  $\frac{f(x + h) - f(x)}{h} = \frac{(x^2 + 2xh + h^2 + 1) - (x^2 + 1)}{h}$ \\
  $\frac{f(x + h) - f(x)}{h} = \frac{2xh + h^2}{h}$ \\
  $\frac{f(x + h) - f(x)}{h} = \frac{h(2x + h)}{h}$ \\
  $\frac{f(x + h) - f(x)}{h} = 2x + h$ \\
\end{example}
