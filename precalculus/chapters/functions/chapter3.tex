\subsection{Finding Domains}
\begin{definition}[Domains]
  Domains is the set of inputs of a function that give a $\R$ number output.
\end{definition}

\subsubsection{The Problem Areas}
\begin{enumerate}
\item \textbf{Square Roots: } $\sqrt{()}$ radicand $> 0$ .
\item \textbf{Denominators: } $\frac{}{()}$ $\ne$ zero.
\end{enumerate}

\begin{example}[Square Root]
  $$f(x) = \sqrt{5 - 4x}$$
  $$5 - 4x \ge 0$$
  $$-4x \ge -5$$
  $$x \ge \frac{-5}{-4}$$
  $$x \le \frac{5}{4}$$
  Can be written as
  $$D = \{x \mid x \le \frac{5}{4}\}$$
  \center{Or}
  $$ (-\infty, \frac{5}{4}]$$
\end{example}

\begin{example}[Denominator]
  $$g(t) = \frac{5t}{t^3 - 16t}$$
  $$t^3 - 16t = 0$$
  $$t(t^2 - 16) = 0$$
  $$t(t - 4)(t + 4)$$
  $$t \ne 0 \lor t \ne 4 \lor t \ne -4$$
  Can be written as
  $$D = \{t \in \R \mid t \ne 0, t \ne -4, t \ne 4 \}$$
  \center{Or}
  $$(-\infty, -4) \cup (-4, 0) \cup (0, 4) \cup (4, \infty)$$
\end{example}

\begin{example}[Examples]
  \hfill
  \begin{enumerate}
  \item $f(x) = 3x^2 - 2x + 4$ \\
    $D: \{x \in \R \}$
    Or
    $(-\infty, \infty)$
  \item $h(x) = \frac{\sqrt{3x - 15}}{x - 4}$ \\
    $x \ne 4$ \\
    $3x - 15 \ge 0$ \\
    $3x \ge 15$ \\
    $x \ge 5$ \\
    $D: \{x \mid x \ne 5\}$ or $[5, \infty)$
  \item $g(x) = \frac{5}{\sqrt{x - 8}}$ \\
    $x - 8 > 0$ \\
    $D: \{ x \mid x > 8\}$ or $(x, \infty)$
  \end{enumerate}
\end{example}
