\subsection{Operations Of Functions}
\begin{definition}[Operations Of Function]
  Function operations invole combining two or more functions to create a new function using standard arithmetic: \textbf{addition, subtraction, multiplication} and \textbf{division}.
\end{definition}
\begin{example}[Operations]
  \hfill
  $$f(x) = \frac{2x + 3}{3x - 2} \qquad g(x) = \frac{4x}{3x - 2}$$
  $$D: \left\{x \mid x \ne \frac{2}{3}\right\} $$
  \begin{enumerate}
  \item $$(f + g)(x) = \frac{2x + 3}{3x - 2} + \frac{4x}{3x - 2} \to \frac{6x + 3}{3x-2} \quad D: \left\{x \mid x \ne \frac{2}{3}\right\}$$
  \item $$(f - g)(x) = \frac{2x + 3}{3x - 2} - \frac{4x}{3x - 2} \to \frac{-2x + 3}{3x - 2} \quad D: \left\{x \mid x \ne \frac{2}{3}\right\}$$
  \item $$(f \cdot g)(x) = \left(\frac{2x + 3}{3x - 2}\right)\left(\frac{4x}{3x - 2}\right) \to \frac{8x^2 + 12x}{(3x - 2)(3x - 2)} \to \frac{8x^2 + 12x}{(3x - 2)^2} \quad D: \left\{x \mid x \ne \frac{2}{3}\right\}$$
  \item $$\left(\frac{f}{g}\right)(x) = \frac{\frac{2x + 3}{3x - 2}}{\frac{4x}{3x - 2}} \to \frac{2x + 3}{3x - 2} \cdot \frac{3x - 2}{4x} \to \frac{2x + 3}{4x} \quad D \left\{x \mid x \ne \frac{2}{3}, x \ne 0\right\}$$
  \end{enumerate}
\end{example}

\begin{example}[Square Root and Fractions]
  \hfill
  $$f(x) = \sqrt{x + 3} \qquad g(x)=\frac{5}{x}$$
  $$D: \{x \mid \ge -3 \} \qquad D: \{x \mid x \ne 0\}$$
  \begin{enumerate}
  \item $$(f + g)(x) = \sqrt{x + 3} + \frac{5}{x} \qquad D: \{x \mid x \ge -3, x \ne 0\}$$
  \item $$(f - g)(x) = \sqrt{x + 3} - \frac{5}{x}\qquad D: \{x \mid x \ge -3, x \ne 0\}$$
  \item $$(f \cdot g)(x) = \frac{5 \sqrt{x + 3}}{x}\qquad D: \{x \mid\ x \ge -3, x \ne 0\}$$
  \item $$\left(\frac{f}{g}\right)(x) = \sqrt{x + 3} \cdot \frac{x}{5} \to \frac{x\sqrt{x + 3}}{5} \qquad D: \{x \mid\ x \ge -3, x \ne 0\}$$
  \end{enumerate}
\end{example}
