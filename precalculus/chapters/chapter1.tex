\section{Functions}
\subsection{Intro To Functions}

\begin{definition}[Function]
A function is a relationsip that maps 1 input to 1 output.
\end{definition}

\textbf{Note: }An input never gives more than 1 output.

A function usually looks like $$ y = f(x) $$ 


$ x $ is the \textit{independent} variable.


\noindent $ y $ or $ f(x) $ is the \textit{Dependent} variable.


\noindent Where the standard ordered pairs looks like $$ (x, y) \lor (x, f(x)) $$

\begin{definition}[Domain]
A domain is a set of inputs.
\end{definition}
\begin{definition}[Range]
A range is a set of outputs.
\end{definition}

\begin{example}[Real Example]
    \hfill
    \begin{table}[H]
    \centering
        \begin{tabular}{cc}
        \toprule
        \textbf{Work Hours} & \textbf{Pay} \\ \midrule
        25 & \$500 \\
        53 & \$310 \\
        30 & \$490 \\
        40 & \$490 \\ \bottomrule
        \end{tabular}
    \end{table}

    \textbf{Note: }This is a function but not a \textit{one-to-one function}.
    
    $Domain: {23, 53, 30, 40}$\\
    $Range: {\$500, \$310, \$490}$
\end{example}

\begin{example}[Input Output]
    $$S = {(-2, 16), (-1, 4), (0, 3), (1, 4)} $$ This is a function.


    $$R = {(-2, 5), (3, 9), (5, 0), (-2, 6)} $$ This is not a function.
\end{example}

\begin{example}[Algebraic]
    $$y = \frac{1}{2}x - 2 $$


    This is a function.

    $$y = 2x^2 - 5x + 4 $$


    This is a function.


    $$y = \pm\sqrt{3 - 2 x}$$


    This is not a function because we get more than 1 output for only 1 input.


    \textbf{Note: } $y^2 = x \ne y = \sqrt{x}$, instead it equals to $y = \pm \sqrt{x}$. Which makes it not a function.
\end{example}

\begin{example}[Factoring]
    $$xy + 2y = 3x - 1$$
    $$y(x + 2) = 3x - 1$$
    $$y = \frac{3x - 1}{x + 2}$$
\end{example}
