\subsection{Factoring Method for Quadratics}
\begin{definition}[Factoring]
	Factoring a quadratic is the process of rewriting a standard quadratic expression $ax^2 + bx + c$ as the product of its linear factors, like $(x - p)(x - 1)$.
\end{definition}
\begin{example}[Factoring]
	\hfill
	\begin{enumerate}
		\item $f(x) = x^2 + 7x + 6$
		\item $g(x) = 3x(x - 4) - 36$
		\item $g(x) = 3x^2 + 2 + 5x$
		\item $h(x) = -2x^2 - 5x + 12$
		\item $f(x) = -12x + 9 + 4x^2$
		\item $g(x) = x^2 - 5$
	\end{enumerate}
\end{example}
