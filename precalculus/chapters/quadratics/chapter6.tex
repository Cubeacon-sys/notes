\subsection{The Quadratic Formula}
\begin{definition}[The Quadratic Formula]
  $$\text{For } ax^2 + bx + c = 0$$
  $$x = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a}$$
  The quadratic formula does the same thing as both the square root method and factoring method. The only difference is that it works all the time.
\end{definition}

\textbf{Note: }Completing the square also works all the time but its harder than the quadratic formula.

\begin{example}[The Quadratic Formula]
  \hfill
  \begin{enumerate}
  \item $f(x) = 3x^2 - 5x + 1$ \\
    $3x^2 - 5x + 1 = 0$ \\
    $x = \frac{-(-5) \pm \sqrt{(-5)^2 - 4 (3)(1)}}{2 (3)}$ \\
    $x = \frac{5 \pm \sqrt{25 - 12}}{6} \to \frac{5 \pm \sqrt{13}}{6}$ \\
    $x = \frac{5 + \sqrt{13}}{6} \land x = \frac{5 - \sqrt{13}}{6}$
  \item $g(x) = 3x(x + 2) - 1$ \\
    $3x^2 + 6x - 1 = 0$ \\
    $x = \frac{-(6) \pm \sqrt{(6)^2 - 4 (3)(-1)}}{2 (3)}$ \\
    $x = \frac{-6 \pm \sqrt{36 + 12}}{6}$ \\
    $x = \frac{-6 \pm \sqrt{48}}{6} \to \frac{-6 \pm {4\sqrt{3}}}{6}$ \\
    $x = \frac{-3 \pm 2\sqrt{3}}{3}$ \\
    $x = \frac{-3 + 2\sqrt{3}}{3} \land x = \frac{-3 - 2\sqrt{3}}{3}$
  \item $h(x) = 3x^2 + x - \frac{1}{2}$ \\
    $3x^2 + x - \frac{1}{2} = 0$ \\
    $2(3x^2 + x - \frac{1}{2}) = 0 \cdot 2$ \\
    $6^2 + 2x - 1 = 0$ \\
    $x = \frac{-(2) \pm \sqrt{(2)^2 - 4 (6)(-1)}}{2 (6)}$ \\
    $x = \frac{-2 \pm \sqrt{4 + 24}}{12}$ \\
    $x = \frac{-2 \pm \sqrt{28}}{12} \to x = \frac{-2 \pm 2\sqrt{7}}{12}$ \\
    $x = \frac{-1 \pm \sqrt{7}}{6}$ \\
    $x = \frac{-1 + \sqrt{7}}{6} \land x = \frac{-1 + \sqrt{7}}{6}$
  \item $f(x) = \frac{1}{8}x^2 - \frac{1}{4}x - 2$ \\
    $\frac{1}{8}x^2 - \frac{1}{4}x - 2 = 0$ \\
    $8(\frac{1}{8}x^2 - \frac{1}{4}x - 2 = 0 \cdot 8)$ \\
    $x^2 - 2x - 16 = 0$ \\
    $x = \frac{-(-2) \pm \sqrt{(-2)^2 - 4 (1)(-16)}}{2 (1)}$ \\
    $x = \frac{2 \pm \sqrt{4 + 64}}{2}$ \\
    $x = \frac{2 \pm \sqrt{68}}{2}$ \\
    $x = \frac{2 \pm 2\sqrt{17}}{2} \to 1 \pm \sqrt{17}$ \\
    $x = 1 + \sqrt{17} \land x = 1 - \sqrt{17}$
  \end{enumerate}
\end{example}

\begin{example}[Special Cases]
  \hfill
  \begin{enumerate}
  \item $f(x) = 9x - 6x + 1$ \\
    $9x - 6x + 1 = 0$ \\
    $x = \frac{-(-6) \pm \sqrt{(-6)^2 - 4 (9)(1)}}{2 (9)}$ \\
    $x = \frac{6 \pm \sqrt{36 - 36}}{18}$ \\
    $x = \frac{6 \pm \sqrt{0}}{18}$ \\
    $ \frac{6 + 0}{18} \land = \frac{6 - 0}{18}$ \\
    $x = \frac{1}{3}$
  \item $g(x) = -4x^2 - x - 4$ \\
    $-4x^2 - x - 4 = 0$ \\
    \textbf{Note: }We can change all the negatives to positives here. \\
    $4x^2 + x + 4 = 0$ \\
    $x = \frac{-(1) \pm \sqrt{(1)^2 - 4 (4)(4)}}{2 (4)}$ \\
    $x = \frac{-1 \pm \sqrt{1 - 64}}{8}$ \\
    $x = \frac{-1 \pm \sqrt{-63}}{8} $ \\
    $x = \frac{-1 \pm 3i\sqrt{7}}{8}$ \\
    $x = \frac{-1 + 3i\sqrt{7}}{8} \land \frac{-1 + 3i\sqrt{7}}{8}$
  \end{enumerate}

\end{example}
